/* ══════ AJAX: Müşteri Kayıt → CargoTrack API'sine gönder ══════ */
add_action('wp_ajax_itmms_register_customer','itmms_ajax_register_customer');
add_action('wp_ajax_nopriv_itmms_register_customer','itmms_ajax_register_customer');
function itmms_ajax_register_customer(){
    $api_url = get_option('itmms_cargotrack_url','');
    $api_key = get_option('itmms_cargotrack_api_key','');
    if(empty($api_url)){wp_send_json_error(['message'=>'Sistem yapılandırması eksik. Lütfen yönetici ile iletişime geçin.']);return;}
    $company = sanitize_text_field($_POST['company_name'] ?? '');
    if(empty($company)){wp_send_json_error(['message'=>'Şirket adı zorunludur.']);return;}
    $fields = ['company_name','physical_address','postal_code','city','country','contact_person','contact_email','contact_phone','tax_number','mersis_number','eori_number','billing_address','billing_email'];
    $body = ['action'=>'ctapi_register_customer','api_key'=>$api_key];
    foreach($fields as $f){
        $body[$f] = sanitize_text_field($_POST[$f] ?? '');
    }
    $endpoint = rtrim($api_url,'/').'/wp-admin/admin-ajax.php';
    $response = wp_remote_post($endpoint,[
        'timeout'   => 15,
        'body'      => $body,
        'sslverify' => false,
    ]);
    if(is_wp_error($response)){
        wp_send_json_error(['message'=>'Bağlantı hatası: '.$response->get_error_message()]);return;
    }
    $code = wp_remote_retrieve_response_code($response);
    $raw_body = wp_remote_retrieve_body($response);
    $data = json_decode($raw_body,true);
    if($data && isset($data['success']) && $data['success']){
        wp_send_json_success($data['data'] ?? ['message'=>'Kayıt başarılı!']);
    } else {
        $msg = isset($data['data']['message']) ? $data['data']['message'] : 'Kayıt sırasında bir hata oluştu.';
        wp_send_json_error(['message'=>$msg]);
    }
}
